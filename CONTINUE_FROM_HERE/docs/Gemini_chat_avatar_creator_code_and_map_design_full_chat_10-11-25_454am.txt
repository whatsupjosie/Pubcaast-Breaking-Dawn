# avatar.py

class Avatar:
    """
    Represents a single avatar character.
    This class holds all the customizable properties.
    """
    def __init__(self, name, core_color="#FFD700", mood_ring_color="#ADFF2F", eyebrow_shape="neutral"):
        """
        Initializes a new Avatar instance.

        Args:
            name (str): The unique name for this avatar.
            core_color (str): The hex color code for the core and features (e.g., gold).
            mood_ring_color (str): The hex color code for the outer ring (e.g., lime green).
            eyebrow_shape (str): A string describing the eyebrow state (e.g., "neutral", "raised", "furrowed").
        """
        self.name = name
        self.core_color = core_color
        self.mood_ring_color = mood_ring_color
        self.eyebrow_shape = eyebrow_shape
        self.accessories = [] # A list to hold accessory names, e.g., ["glasses", "hat"]
        self.hair_style = None # A string for the hair style name, e.g., "spiky"

    def change_mood_color(self, new_color):
        """Changes the color of the mood ring."""
        print(f"Changing {self.name}'s mood ring color to {new_color}")
        self.mood_ring_color = new_color

    def add_accessory(self, item):
        """Adds an accessory to the avatar."""
        if item not in self.accessories:
            print(f"Adding {item} to {self.name}")
            self.accessories.append(item)
        else:
            print(f"{self.name} already has {item}.")

    def __repr__(self):
        """A simple string representation of the avatar for printing."""
        return f"Avatar(Name: {self.name}, Core: {self.core_color}, Mood: {self.mood_ring_color})"

# --- Example of how to use this class ---
# Create a new avatar instance based on our template
base_avatar = Avatar(name="Alpha")

# Customize it
base_avatar.change_mood_color("#FF4500") # Change mood ring to orange-red
base_avatar.add_accessory("sunglasses")
base_avatar.eyebrow_shape = "raised"

# Print the object to see its current state
print(base_avatar)
print(f"Accessories: {base_avatar.accessories}")
print(f"Eyebrow Shape: {base_avatar.eyebrow_shape}")

  canvas: document.querySelector('#mirror-canvas'),

  alpha: true,     // Allows for a transparent background

// A scene is the container for all 3D objects.
const scene = new THREE.Scene();

// A camera defines our point of view.
const camera = new THREE.PerspectiveCamera(75, 1, 0.1, 1000);
camera.position.z = 5; // Move the camera back so we can see the objects

// --- Avatar Object ---
// Let's create a simple sphere to represent the avatar's head.
const geometry = new THREE.SphereGeometry(2, 32, 32); // Radius 2, with 32 segments for smoothness

// This material will give it a basic color. We'll make it glow later.
const material = new THREE.MeshBasicMaterial({ color: 0xffd84d }); // A gold-like color
const avatarSphere = new THREE.Mesh(geometry, material);
scene.add(avatarSphere); // Add the sphere to the scene

// --- Animation Loop ---
function animate() {
  // Request the browser to run this function again on the next frame
  requestAnimationFrame(animate);

  // Simple animation: rotate the avatar sphere
  avatarSphere.rotation.y += 0.005;

  // Render the scene from the camera's perspective
  renderer.render(scene, camera);
}

// static/dressing.js
import * as THREE from 'three';

// --- Shader Code ---

const vertexShader = `
  varying vec3 vNormal;
  varying vec2 vUv;
  void main() {
    vUv = uv;
    vNormal = normalize(normalMatrix * normal);
    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
  }
`;

const fragmentShader = `
  uniform vec3 uCoreColor;
  uniform vec3 uMoodRingColor;
  varying vec3 vNormal;
  varying vec2 vUv;
  uniform float uTime;

  // Function to create a procedural line pattern
  float drawLines(vec2 p, float frequency, float thickness) {
    float wave1 = sin(p.x * frequency + uTime * 2.0) * 0.5 + 0.5;
    float wave2 = sin(p.y * frequency * 1.5 + uTime * 1.5) * 0.5 + 0.5;
    float combined = wave1 * wave2;
    return smoothstep(thickness, thickness + 0.1, combined);
  }

  void main() {
    // --- Rim Lighting Effect (Fresnel) ---
    // This creates the glow on the edges of the sphere.
    float fresnel = 1.0 - dot(vNormal, vec3(0.0, 0.0, 1.0));
    fresnel = pow(fresnel, 2.0); // Power to control the falloff

    // --- Mood Ring Border ---
    // Make the fresnel effect sharper at the very edge.
    float moodRingIntensity = smoothstep(0.65, 1.0, fresnel);

    // --- Core Woven Texture ---
    // We use the UV coordinates to create the woven pattern.
    float linePattern = drawLines(vUv * 20.0, 5.0, 0.4);
    vec3 coreTexture = uCoreColor * linePattern;

    // --- Combine Layers ---
    // Start with the core texture
    vec3 finalColor = coreTexture;

    // Additive blending for the fresnel glow effect on top of the core
    finalColor += uCoreColor * fresnel * 0.5;

    // Mix the mood ring color at the very edge, based on its intensity
    finalColor = mix(finalColor, uMoodRingColor, moodRingIntensity);

    gl_FragColor = vec4(finalColor, moodRingIntensity + fresnel * 0.5 + linePattern * 0.8);
  }
`;

// --- Scene Setup ---
const renderer = new THREE.WebGLRenderer({
  canvas: document.querySelector('#mirror-canvas'),
  antialias: true,
  alpha: true,
});
renderer.setSize(600, 600);
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(75, 1, 0.1, 1000);
camera.position.z = 4;

// --- Avatar Object with ShaderMaterial ---
const geometry = new THREE.SphereGeometry(2, 64, 64); // Increased segments for smoother fresnel

// Uniforms are variables we pass from our JavaScript to the shaders.
const uniforms = {
  uCoreColor: { value: new THREE.Color(0xffd84d) },
  uMoodRingColor: { value: new THREE.Color(0xadff2f) },
  uTime: { value: 0.0 }
};

const material = new THREE.ShaderMaterial({
  vertexShader: vertexShader,
  fragmentShader: fragmentShader,
  uniforms: uniforms,
  transparent: true, // MUST be true for alpha blending to work
  blending: THREE.AdditiveBlending, // Makes the glows look better
});

const avatarSphere = new THREE.Mesh(geometry, material);
scene.add(avatarSphere);

const clock = new THREE.Clock(); // Clock to keep track of time for animations

// --- Animation Loop ---
function animate() {
  requestAnimationFrame(animate);

  // Update the time uniform for the shader animation
  uniforms.uTime.value = clock.getElapsedTime();

  avatarSphere.rotation.y += 0.005;
  renderer.render(scene, camera);
}

// --- Fetch Avatar Data and Start ---
async function initialize() {
  try {
    // This endpoint now comes from the server.py file we just created.
    // Let's get the default 'alpha_base' character.
    const response = await fetch('/api/characters/alpha_base');
    const avatarData = await response.json();

    console.log('Fetched avatar data:', avatarData);

    // Update the shader uniforms with the colors from our Python backend
    uniforms.uCoreColor.value.set(avatarData.primary_color);
    // For now, let's hardcode the mood ring color, we can add it to the backend later.
    // uniforms.uMoodRingColor.value.set(avatarData.mood_ring_color);

    animate();
  } catch (error) {
    console.error('Failed to load avatar data:', error);
  }

Overview of the System
This code is designed to create a system where you can use different AI models (like those from OpenAI, Google Gemini, and Anthropic) through a common interface. The core idea is to abstract away the specific details of each AI provider's API.

base.py defines the framework and the rules.

The ..._adapter.py files are plugins or drivers that follow those rules for a specific AI service.

## base.py: The Foundation
This is the most important file as it defines the core components and structure for the entire agent system.

ModelAdapter: This is a Protocol (a blueprint for a class). It declares that any class acting as a model adapter must have a stream_reply method. This ensures that the main application code can talk to any AI model in the exact same way, without needing to know if it's Gemini or OpenAI.

AgentRuntime: This class is a stateful wrapper around an agent's configuration and its adapter. It's responsible for tracking things like:

When the agent was last used (last_invoked).

Whether the agent is ready to be used again based on its cooldown period (ready()).

How many times its API calls have failed (failures).

AdapterFactory: This class is responsible for creating the correct ModelAdapter instance based on an agent's configuration. You register different "builders" (like OpenAIAdapter or GeminiAdapter) with it. When you need an adapter, you give it the agent's config, and it returns the right one.

AgentRegistry: This acts as a catalog of all available AI agents you've configured. It holds their AgentConfig and uses the AdapterFactory to create a live AgentRuntime instance when needed.

## The Adapters: Connecting to AI Services
These files are concrete implementations of the ModelAdapter protocol. Each one knows how to communicate with a specific AI provider's API.

openai_adapter.py
This file allows your system to connect to OpenAI models.

Initialization: It reads the OpenAI API key from an environment variable specified in the agent's configuration (api_key_env).

stream_reply method: It constructs an HTTP request payload specifically formatted for the OpenAI API, including the model name and prompt.

Streaming: It makes a POST request to the OpenAI API endpoint (https://api.openai.com/v1/responses) and processes the server-sent events line by line, yielding each piece of text as a Delta object.

gemini_adapter.py
This adapter connects to Google's Gemini models.

Initialization: Similar to the OpenAI adapter, it securely loads the Gemini API key from an environment variable.

stream_reply method: It builds a request payload that matches the structure required by the Gemini API ("contents": [{"role": "user", ...}]).

Streaming: It sends the request to the Gemini API endpoint and processes the streaming JSON response, extracting the text from the candidates field in the response data.

anthropic_adapter.py

Initialization: It follows the same secure pattern of loading an API key from an environment variable.

stream_reply method: It formats the request for the Anthropic Messages API, including the required headers like anthropic-version.

Streaming: It connects to the Anthropic API endpoint (https://api.anthropic.com/v1/messages) and processes the streaming data, looking for lines that start with "data:" to parse the JSON chunks.

How It All Works Together
You define an agent's settings in an AgentConfig object, specifying its provider as "openai", "gemini", etc.

The AgentRegistry holds this configuration.

When the application needs to use this agent, it asks the AgentRegistry for an AgentRuntime instance.

The registry uses the AdapterFactory to build the correct adapter (e.g., GeminiAdapter) for that agent.

The application can then call the stream_reply method on the adapter, and the specific adapter handles all the provider-specific communication, returning the AI's response in a standardized way.

How the Code Implements Your Vision
Our system is broken into a few key parts that match your idea of having a primary look that can be customized.

1. The Base Template (The 3D Sphere)
The fundamental "body" of your avatar is created in the dressing.js file.

The Shape: The line const geometry = new THREE.SphereGeometry(2, 64, 64); creates the perfect sphere. This is the blank canvas for every avatar.

The Look: The custom shader code we wrote is what gives the sphere its signature "neon scribble" texture and glow. This is the digital equivalent of your ALPHA AVATAR PROTO TYPE STILL.jpg image.

This combination of a sphere and a special shader is the base template for every single avatar.

2. Changing the Color
The base template can be any color you want.

In the backend (modules/avatar.py), the AvatarState class has a primary_color property.

In the frontend (dressing.js), the shader has a variable uColor that receives this color.

When you pick a new "Aura Color" in the wardrobe, the JavaScript simply tells the shader to use that new color for the base sphere. This allows the primary look to be yellow, blue, pink, or anything else.

3. Adding Customizations (The Layers)
The hairstyles, glasses, and other accessories you've shown in the example images are treated as separate pieces that we layer on top of the base sphere.

Data Storage: In modules/avatar.py, the features and accessories dictionaries in the AvatarState are where we keep track of what's been added (e.g., features: {'hair': 'ponytail'} or accessories: {'glasses': 'round'}).

Visual Application: In the frontend, the renderAvatarCSS() function in dressing.js reads these dictionaries. If it sees that you have 'hair': 'ponytail', it applies the .hair.ponytail CSS class to the HTML. This makes the ponytail style appear as an overlay on top of the base 3D avatar, just like putting a hat on a mannequin.

This modular approach is incredibly flexible. We can add hundreds of accessories and hairstyles just by defining new CSS rules for them, without ever having to change the core 3D avatar. We are perfectly set up to build out all the variations you've imagined.

  // Uniforms: Variables passed from JavaScript
  uniform vec3 uCoreColor;
  uniform vec3 uMoodRingColor;
  uniform float uTime; // Time for animation

  // Varyings: Variables passed from the vertex shader
  varying vec3 vNormal;
  varying vec2 vUv;

  // Function to create a procedural, swirling line pattern
  // This simulates the "woven" or "scribble" texture.
  float noise(vec2 p) {
    return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
  }

  float map(vec2 p, float scale) {
    p *= scale;
    float n = noise(p);
    float n2 = noise(p + n);
    return sin(n * 3.14159 * 2.0 + sin(n2 * 3.14159 * 2.0 + uTime * 1.5));
  }

  void main() {
    // --- Layer 1: The Mood Ring (Outer Edge) ---
    // The "fresnel" effect is strongest on the sphere's edges.
    float fresnel = 1.0 - dot(vNormal, vec3(0.0, 0.0, 1.0));
    float moodRingIntensity = pow(fresnel, 4.0); // Use a high power for a sharp edge

    // --- Layer 2: The Woven Core Texture ---
    // Use the UV coordinates to create the scribble pattern.
    float wovenPattern = map(vUv, 20.0) * 0.5 + 0.5;

    // This is another fresnel, but softer, to make the center feel lit.
    float innerGlowIntensity = pow(fresnel, 2.0) * 0.5;

    float alpha = wovenPattern * 0.7 + innerGlowIntensity + moodRingIntensity;

    gl_FragColor = vec4(finalColor, alpha);
  }

def _avatar_path(data_dir, user_id: str):
    return data_dir / "avatars" / f"{user_id}.json"

def load_avatar(data_dir, user_id: str) -> AvatarState:
    path = _avatar_path(data_dir, user_id)
    data = read_json(path)
    if not data:
        return default_avatar()
    try:
        # Ensure accessories and features are present
        data.setdefault("accessories", {})
        data.setdefault("features", {})
        return AvatarState(**data)
    except ValidationError as exc:
        logger.warning("Invalid avatar state for %s: %s. Resetting.", user_id, exc)
        return default_avatar()

def save_avatar(data_dir, user_id: str, payload: Dict) -> AvatarState:
    path = _avatar_path(data_dir, user_id)
    path.parent.mkdir(parents=True, exist_ok=True)
    current = load_avatar(data_dir, user_id).dict()
    
    # Deep merge accessories and features
    current_accessories = current.get("accessories", {})
    payload_accessories = payload.get("accessories", {})
    current_accessories.update(payload_accessories)
    payload["accessories"] = current_accessories

    current_features = current.get("features", {})
    payload_features = payload.get("features", {})
    current_features.update(payload_features)
    payload["features"] = current_features

    current.update(payload)
    state = AvatarState(**current)
    write_json(path, state.dict())
    return state

  uniform vec3 uCoreColor;
  uniform vec3 uMoodRingColor;

  // --- Procedural Noise Function ---
  // This function creates a pseudo-random value, which is the
  // foundation for our swirling, "woven" texture.
  float noise(vec2 p) {
    return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
  }

  // --- Swirling Pattern Function ---
  // This combines multiple layers of noise and animates them with uTime
  // to create the organic, fluid motion of the woven light strands.
  float swirlingPattern(vec2 p, float scale) {
    p *= scale;
    float n1 = noise(p + uTime * 0.1);
    float n2 = noise(p + n1 + uTime * 0.2);
    return sin(p.x * n1 + p.y * n2);

    // The "fresnel" effect calculates how much a surface is angled away
    // from the camera. It's strongest at the very edges of the sphere.

def save_avatar(data_dir, user_id: str, payload: Dict) -> AvatarState:
    path = _avatar_path(data_dir, user_id)
    path.parent.mkdir(parents=True, exist_ok=True)

    # Deep merge accessories and features
    current_accessories = current.get("accessories", {})
    payload_accessories = payload.get("accessories", {})

    payload_features = payload.get("features", {})
    current_features.update(payload_features)
    payload["features"] = current_features

    current.update(payload)
    state = AvatarState(**current)
    write_json(path, state.dict())

// Main game loop
function update() {
    if (!isMovementPaused) {
        handleInput();
    }
    draw();
    requestAnimationFrame(update);

// Handle Keyboard Input
window.addEventListener('keydown', e => keysPressed[e.code] = true);
window.addEventListener('keyup', e => keysPressed[e.code] = false);

function handleInput() {
    let dx = 0, dy = 0;
    if (keysPressed['ArrowUp'] || keysPressed['KeyW']) dy = -1;
    if (keysPressed['ArrowDown'] || keysPressed['KeyS']) dy = 1;
    if (keysPressed['ArrowLeft'] || keysPressed['KeyA']) dx = -1;
    if (keysPressed['ArrowRight'] || keysPressed['KeyD']) dx = 1;
    if (dx !== 0 || dy !== 0) movePlayer(dx, dy);
}

// Player Movement and Interaction
function movePlayer(dx, dy) {
    const nextX = player.x + dx * player.speed;
    const nextY = player.y + dy * player.speed;
    const gridX = Math.floor(nextX / TILE_SIZE);
    const gridY = Math.floor(nextY / TILE_SIZE);

    if (navMesh[gridY] && navMesh[gridY][gridX] === 0) {
        player.x = nextX;
        player.y = nextY;
    }

    // After moving, check for trigger activations
    for (const zone of triggerZones) {
        if (player.x > zone.x && player.x < zone.x + zone.width && player.y > zone.y && player.y < zone.y + zone.height) {
            if (zone.type === 'confirm') showConfirmationPrompt(zone);
            return;
        }
    }
}

// Confirmation Pop-up
function showConfirmationPrompt(zone) {
    if (document.querySelector('.confirmation-modal')) return;

                <button id="confirm-no">No</button>
            </div>
        </div>`;

// Main draw function that renders the entire scene
function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    const bg = loadedImages[currentView];
    if (bg && bg.complete) {
        ctx.drawImage(bg, 0, 0, canvas.width, canvas.height);
    }
    
    drawInteractionCues();
    drawPlayer();
}

// Draw the player avatar sprite
function drawPlayer() {
    // This function renders a simple "neon scribble" style avatar.
    // It will be replaced by your full avatar system later.
    const glowColor = '#00ffff'; // A bright cyan
    ctx.save(); // Save the current state of the canvas

    // Create a glowing effect
    ctx.strokeStyle = glowColor;
    ctx.fillStyle = glowColor;
    ctx.shadowColor = glowColor;
    ctx.shadowBlur = 20;
    
    // Draw the head (a simple circle)
    ctx.beginPath();
    ctx.arc(player.x, player.y - player.height / 2, player.width / 2, 0, Math.PI * 2);
    ctx.stroke();

    // Draw the eyes
    ctx.beginPath();
    ctx.arc(player.x - 10, player.y - 70, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(player.x + 10, player.y - 70, 5, 0, Math.PI * 2);
    ctx.fill();

    // Draw the mouth (a simple smile)
    ctx.beginPath();
    ctx.arc(player.x, player.y - 55, 15, 0.2 * Math.PI, 0.8 * Math.PI);
    ctx.stroke();

    ctx.restore(); // Restore the canvas state
}

// Draw visual cues for interactive zones
function drawInteractionCues() {
    for (const zone of triggerZones) {
        // Check if the player is near the zone
        const isNear = Math.hypot(player.x - (zone.x + zone.width/2), player.y - (zone.y + zone.height/2)) < 200;

        if (isNear) {
            ctx.save();
            ctx.strokeStyle = '#e6b91e'; // Gold color
            ctx.lineWidth = 4;
            ctx.shadowColor = '#e6b91e';
            ctx.shadowBlur = 20;
            // Draw a subtle, glowing rectangle around the interactive zone
            ctx.strokeRect(zone.x, zone.y, zone.width, zone.height);
            ctx.restore();
        }
    }
}
